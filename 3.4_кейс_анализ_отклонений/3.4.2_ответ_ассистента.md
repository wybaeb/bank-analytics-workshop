```python
# Ячейка 1: Импорт библиотек

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from datetime import timedelta


# Ячейка 2: Чтение файла и подготовка данных

df = pd.read_csv('data/series/portfolio_operations_daily.csv', sep=';', decimal=',', parse_dates=['дата'], index_col='дата')
df.index.freq = 'D'
volume_series = df['объём_операций_млн_руб']
monthly_volume = volume_series.resample('M').sum()


# Ячейка 3: График объёма операций с линейным трендом и продолжением на полгода вперед

plt.figure(figsize=(12, 6))
X = monthly_volume.reset_index().index.values.reshape(-1, 1)
y = monthly_volume.values.reshape(-1, 1)
model = LinearRegression().fit(X, y)
future_X = np.arange(len(monthly_volume), len(monthly_volume)+6).reshape(-1, 1)
future_y = model.predict(future_X)
plt.plot(monthly_volume.index, monthly_volume, label="Факт")
plt.plot(pd.date_range(start=monthly_volume.index[-1]+timedelta(days=1), periods=6, freq='M'), future_y.flatten(), '--r', label="Прогноз")
plt.legend()
plt.show()


# Ячейка 4: Расчёт и визуализация сезонности

weekly_seasonality = volume_series.groupby(volume_series.index.dayofweek).mean() / volume_series.mean()
intra_month_seasonality = volume_series.groupby(volume_series.index.day).mean() / volume_series.mean()
yearly_seasonality = volume_series.groupby(volume_series.index.month).mean() / volume_series.mean()

fig, axes = plt.subplots(1, 3, figsize=(15, 5))
axes[0].bar(range(7), weekly_seasonality)
axes[0].set_title("Недельная сезонность")
axes[1].plot(intra_month_seasonality)
axes[1].set_title("Внутримесячная сезонность")
axes[2].bar(range(1, 13), yearly_seasonality)
axes[2].set_title("Годовая сезонность")
plt.tight_layout()
plt.show()


# Ячейка 5: Очистка от сезонности и построение графика очищенных данных с трендом последних 12 месяцев

def remove_seasonality(ts):
    ts_week_adj = ts / weekly_seasonality[ts.index.dayofweek]
    ts_intra_month_adj = ts_week_adj / intra_month_seasonality[ts_week_adj.index.day]
    return ts_intra_month_adj / yearly_seasonality[ts_intra_month_adj.index.month]

cleaned_volume = remove_seasonality(volume_series)
cleaned_monthly_volume = cleaned_volume.resample('M').sum()

last_year_data = cleaned_monthly_volume.tail(12)
X_last_year = last_year_data.reset_index().index.values.reshape(-1, 1)
y_last_year = last_year_data.values.reshape(-1, 1)
trend_model = LinearRegression().fit(X_last_year, y_last_year)
future_trend = trend_model.predict(np.arange(12, 18).reshape(-1, 1))

plt.figure(figsize=(12, 6))
plt.plot(cleaned_monthly_volume.index, cleaned_monthly_volume, label="Очищенные данные")
plt.plot(pd.date_range(start=cleaned_monthly_volume.index[-1]+timedelta(days=1), periods=6, freq='M'), future_trend.flatten(), '--r', label="Тренд последних 12 мес.")
plt.legend()
plt.show()


# Ячейка 6: Реализация функции экспоненциального сглаживания Холта

def holt_exponential_smoothing(data, alpha=0.2, beta=0.1):
    level = [data.iloc[0]]
    trend = [data.diff().dropna().iloc[0]]
    
    for i in range(1, len(data)):
        new_level = alpha * data[i] + (1-alpha)*(level[-1] + trend[-1])
        new_trend = beta*(new_level-level[-1])+(1-beta)*trend[-1]
        level.append(new_level)
        trend.append(new_trend)
        
    forecast = []
    for t in range(len(level)-1, len(level)+len(future_trend)):
        forecast.append(level[-1] + trend[-1]*((t+1)-(len(level))))
    return forecast[:len(future_trend)]


# Ячейка 7: Прогнозирование методом скользящего окна и расчёт средней ошибки моделей

errors = {'Linear Trend': [], 'Seasonal Naive': [], 'Holt Exponential': []}
for start_date in pd.date_range(end=pd.to_datetime('today')-pd.DateOffset(years=1), periods=12, freq='MS'):
    train = cleaned_monthly_volume[start_date-pd.DateOffset(years=1):start_date]
    test = cleaned_monthly_volume[start_date:start_date+pd.DateOffset(months=3)]
    
    # Линейный тренд
    X_train = train.reset_index().index.values.reshape(-1, 1)
    y_train = train.values.reshape(-1, 1)
    lin_reg = LinearRegression().fit(X_train, y_train)
    pred_lin = lin_reg.predict(np.arange(len(train), len(train)+3).reshape(-1, 1)).flatten()
    errors['Linear Trend'].append(abs(pred_lin-test.values).mean())
    
    # Seasonal naive
    seasonal_naive_pred = train.shift(12)[-3:]
    errors['Seasonal Naive'].append(abs(seasonal_naive_pred-test.values).mean())
    
    # Holt exponential smoothing
    holt_forecast = holt_exponential_smoothing(train, alpha=0.2, beta=0.1)
    errors['Holt Exponential'].append(abs(holt_forecast-test.values).mean())

error_df = pd.DataFrame(errors)
print(error_df.describe().loc['mean'])


# Ячейка 8: Выбор лучшей модели и построение финального прогноза с доверительными границами

best_model = error_df.mean(axis=0).idxmin()
if best_model == 'Linear Trend':
    final_forecast = trend_model.predict(np.arange(len(cleaned_monthly_volume), len(cleaned_monthly_volume)+6).reshape(-1, 1))
elif best_model == 'Seasonal Naive':
    final_forecast = cleaned_monthly_volume.shift(12).tail(6)
else:
    final_forecast = holt_exponential_smoothing(cleaned_monthly_volume, alpha=0.2, beta=0.1)

std_dev = abs(final_forecast - cleaned_monthly_volume.tail(6)).std()
confidence_interval = std_dev*1.96*np.sqrt(6)
lower_bound = final_forecast - confidence_interval
upper_bound = final_forecast + confidence_interval

plt.figure(figsize=(12, 6))
plt.plot(cleaned_monthly_volume.index, cleaned_monthly_volume, label="Фактические значения")
plt.plot(pd.date_range(start=cleaned_monthly_volume.index[-1]+timedelta(days=1), periods=6, freq='M'), final_forecast, '--b', label=f"{best_model}")
plt.fill_between(pd.date_range(start=cleaned_monthly_volume.index[-1]+timedelta(days=1), periods=6, freq='M'),
                 lower_bound,
                 upper_bound,
                 color='gray',
                 alpha=0.3,
                 label="Доверительный интервал")
plt.legend()
plt.show()
```

**Примечание:** 
- В ячейке №7 используется простая реализация метода Холта без библиотеки `statsmodels`.
- Доверительные границы рассчитываются упрощённо через стандартное отклонение прогнозируемых значений относительно реальных данных за последний период.
