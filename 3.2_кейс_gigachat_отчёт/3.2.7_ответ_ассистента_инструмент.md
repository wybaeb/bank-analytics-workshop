// Основные переменные
let DATA = [], CHARTS = {}, COLUMN_TYPES = {};

// Функция загрузки файла
document.getElementById('fileInput').addEventListener('change', function(event) {
    const file = event.target.files[0];
    if (!file || !/\.(csv|txt)$/.test(file.name)) return alert("Пожалуйста, выберите корректный CSV-файл.");

    let reader = new FileReader();
    reader.onload = () => {
        DATA = processFile(reader.result);
        renderDashboard(DATA);
    };
    reader.readAsText(file);
});

function processFile(content) {
    const rows = content.split("\n").filter(r => r.trim());
    const headers = rows.shift().split(";");
    const result = [];

    rows.forEach(row => {
        const values = row.split(";");
        const obj = {};
        headers.forEach((header, index) => {
            obj[header.trim()] = values[index].trim();
        });
        result.push(obj);
    });

    // Определение типа колонок
    COLUMN_TYPES = detectColumnTypes(result);
    result.forEach(item => {
        for (let col in COLUMN_TYPES) {
            if (COLUMN_TYPES[col] === 'number') {
                item[col] = Number(item[col].replace(',', '.'));
            } else if (COLUMN_TYPES[col] === 'date') {
                item[col] = new Date(item[col]);
            }
        }
    });

    return result;
}

// Автоматическое определение типов колонок
function detectColumnTypes(data) {
    const types = {};
    const firstRow = data[0];
    for (let col of Object.keys(firstRow)) {
        if (/^(\d{4}-\d{2})$/i.test(firstRow[col])) {
            types[col] = 'date'; // Месяцы формата ГГГГ-MM
        } else if (!isNaN(Number(firstRow[col].replace(',', '.')))) {
            types[col] = 'number';
        } else {
            types[col] = 'string';
        }
    }
    return types;
}

// Проверка наличия пропусков
function checkMissingValues(data) {
    const missingCounts = {};
    for (let col of Object.keys(data[0])) {
        missingCounts[col] = data.filter(row => isEmptyOrNull(row[col])).length;
    }
    return missingCounts;
}

// Форматируем даты

    return dateObj instanceof Date ? dateObj.toISOString().substr(0, 7) : dateObj;
}

// Функция проверки пустых значений
function isEmptyOrNull(value) {
    return value === "" || value === undefined || value === null;

// Строим корреляционную матрицу
function buildCorrelationMatrix(data) {
    const mode = document.getElementById('corrmodeSelect').value;
    const numericCols = Object.keys(COLUMN_TYPES).filter(col => COLUMN_TYPES[col] === 'number');
    
    if (mode === 'between') {
        const matrix = computeCorrelationMatrix(data, numericCols);
        renderCorrelationMatrix(matrix, numericCols);
    } else if (mode === 'by_slice') {
        const sliceCols = Object.keys(COLUMN_TYPES).filter(col => COLUMN_TYPES[col] === 'string');
        
        // Для каждого среза вычисляем сводную таблицу и корреляции
        sliceCols.forEach(sliceCol => {
            const groupedData = groupByMonthAndSlice(data, sliceCol);
            
            // Формируем список уникальных метрик
            const metrics = Object.keys(groupedData[0]).filter(col => COLUMN_TYPES[col] === 'number');
            
            // Создаем корреляционные матрицы для каждого среза отдельно
            const matrix = computeCorrelationMatrix(groupedData, metrics);
            renderCorrelationMatrix(matrix, metrics);
        });
    }
}

// Агрегация данных по месяцам и срезам
function groupByMonthAndSlice(data, sliceCol) {
    const aggregatedData = [];
    const months = [...new Set(data.map(d => d['месяц']))];
    const slices = [...new Set(data.map(d => d[sliceCol]))];

    months.forEach(month => {
        slices.forEach(slice => {
            const filteredRows = data.filter(d => d['месяц'] === month && d[sliceCol] === slice);
            const sumMetrics = {};
            Object.keys(filteredRows[0]).forEach(col => {
                if (COLUMN_TYPES[col] === 'number') {
                    sumMetrics[col] = filteredRows.reduce((acc, curr) => acc + curr[col], 0);
                }
            });
            aggregatedData.push(Object.assign({}, {'месяц': month, [sliceCol]: slice}, sumMetrics));
        });
    });

    return aggregatedData;
}

// Класс ячейки корреляционной матрицы
function getCorrelationClass(correlation) {
    if (Math.abs(correlation) >= 0.7) return correlation > 0 ? 'positive' : 'negative';
    if (Math.abs(correlation) >= 0.3) return 'neutral';
    return '';
}

// Вычисление корреляционной матрицы
function computeCorrelationMatrix(data, columns) {
    const n = columns.length;
    const matrix = Array.from({ length: n }, () => Array(n).fill(0));

    for (let i = 0; i < n; i++) {
        for (let j = i; j < n; j++) {
            const x = data.map(row => row[columns[i]]).filter(val => val != null);
            const y = data.map(row => row[columns[j]]).filter(val => val != null);
            matrix[i][j] = pearsonCorrelation(x, y);
            matrix[j][i] = matrix[i][j]; // Симметричная матрица
        }
    }
    return matrix;
}

// Коэффициент Пирсона
function pearsonCorrelation(x, y) {
    const meanX = average(x), meanY = average(y);
    const numerator = x.map((_, i) => (x[i] - meanX) * (y[i] - meanY)).reduce((a,b) => a+b, 0);
    const denominator = Math.sqrt(
        x.map(xi => Math.pow(xi - meanX, 2)).reduce((a,b) => a+b, 0) *
        y.map(yi => Math.pow(yi - meanY, 2)).reduce((a,b) => a+b, 0)
    );
    return denominator === 0 ? 0 : numerator / denominator;
}

// Средняя величина массива
function average(arr) {
    return arr.reduce((a,b) => a+b, 0) / arr.length;
}

// Поиск сильнейших связей
function findStrongestCorrelations(data) {
    const numericCols = Object.keys(COLUMN_TYPES).filter(col => COLUMN_TYPES[col] === 'number');
    const correlations = computeCorrelationMatrix(data, numericCols);

    const strongestPairs = [];
    for (let i = 0; i < numericCols.length; i++) {
        for (let j = i+1; j < numericCols.length; j++) {
            const corr = correlations[i][j];
            strongestPairs.push({
                pair: `${numericCols[i]}, ${numericCols[j]}`,
                strength: Math.abs(corr),
                coefficient: corr
            });
        }
    }

    strongestPairs.sort((a,b) => b.strength - a.strength);

    const listEl = document.getElementById('strongestCorrelationsList');
    listEl.innerHTML = '';
    strongestPairs.slice(0, 5).forEach(pair => {
        const li = document.createElement('li');
        li.textContent = `${pair.pair}: коэффициент=${pair.coefficient.toFixed(2)}, сила=${
            pair.strength >= 0.7 ? 'сильная' :
            pair.strength >= 0.3 ? 'умеренная' : 'слабая'}`;
        listEl.appendChild(li);
    });
}

// Поиск выбросов
function findOutliers(data) {
    const numericCols = Object.keys(COLUMN_TYPES).filter(col => COLUMN_TYPES[col] === 'number');
    const sliceCols = Object.keys(COLUMN_TYPES).filter(col => COLUMN_TYPES[col] === 'string');

    const outliers = [];
    sliceCols.forEach(sliceCol => {
        const uniqueSlices = [...new Set(data.map(row => row[sliceCol]))];
        uniqueSlices.forEach(uniqueVal => {
            numericCols.forEach(numericCol => {
                const subset = data.filter(row => row[sliceCol] === uniqueVal);
                const values = subset.map(row => row[numericCol]).filter(val => val != null);
                const q1 = quantile([...values], 0.25), q3 = quantile([...values], 0.75);
                const iqr = q3 - q1;
                const lowerBound = q1 - 1.5*iqr, upperBound = q3 + 1.5*iqr;
                const median = quantile([...values], 0.5);

                subset.forEach(row => {
                    const val = row[numericCol];
                    if ((val < lowerBound || val > upperBound) && val != null) {
                        outliers.push({
                            slice: sliceCol,
                            sliceValue: uniqueVal,
                            metric: numericCol,
                            month: formatDate(row['месяц']),
                            value: val,
                            deviationFactor: Math.abs((val - median)/median)
                        });
                    }
                });
            });
        });
    });

    outliers.sort((a,b) => b.deviationFactor - a.deviationFactor);

    const tbody = document.getElementById('outliersTable').querySelector('tbody');
    tbody.innerHTML = '';
    outliers.slice(0, 20).forEach(outlier => {
        const tr = document.createElement('tr');
        ['slice', 'sliceValue', 'metric', 'month', 'value', 'deviationFactor'].forEach(key => {
            const td = document.createElement('td');
            td.textContent = key === 'deviationFactor' 
                             ? `${outlier[key].toFixed(2)}×`
                             : outlier[key];
            tr.appendChild(td);
        });
        tbody.appendChild(tr);
    });

    if (outliers.length > 20) {
        document.getElementById('showAllButton').style.display = 'block';
        document.getElementById('showAllButton').onclick = () => {
            tbody.innerHTML = '';
            outliers.forEach(outlier => {
                const tr = document.createElement('tr');
                ['slice', 'sliceValue', 'metric', 'month', 'value', 'deviationFactor'].forEach(key => {
                    const td = document.createElement('td');
                    td.textContent = key === 'deviationFactor'
                                     ? `${outlier[key].toFixed(2)}×`
                                     : outlier[key];
                    tr.appendChild(td);
                });
                tbody.appendChild(tr);
            });
        };
    } else {
        document.getElementById('showAllButton').style.display = 'none';
    }
}

// Квантили
function quantile(arr, p) {
    const sortedArr = [...arr].sort((a,b) => a-b);
    const pos = (sortedArr.length - 1) * p;
    const base = Math.floor(pos);
    const rest = pos - base;
    if (base + 1 < sortedArr.length) {
        return sortedArr[base] + rest * (sortedArr[base + 1] - sortedArr[base]);
    } else {
        return sortedArr[base];
    }
}
