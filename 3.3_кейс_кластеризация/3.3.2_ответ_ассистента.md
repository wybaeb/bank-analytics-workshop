```python
# Ячейка №1: Импорт библиотек

import pandas as pd
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import plotly.express as px


# Ячейка №2: Загрузка данных и просмотр статистики

df = pd.read_csv('data/clients/clients_sample.csv', sep=';', decimal=',', encoding='utf-8')
print(df.describe())
print("Средние значения по всей базе:")
print(df.mean())


# Ячейка №3: Нормализация признаков (StandardScaler)

scaler = StandardScaler()
scaled_data = scaler.fit_transform(df.drop(columns=['id_клиента']))

# Комментарий: нормирование важно, чтобы признаки были сопоставимы друг с другом,
# иначе переменные с большими значениями будут доминировать над остальными.


# Ячейка №4: Выбор числа кластеров методом локтя

inertia_values = []
for k in range(1, 9):
    model = KMeans(n_clusters=k, random_state=42).fit(scaled_data)
    inertia_values.append(model.inertia_)
    
plt.figure(figsize=(8, 5))
sns.lineplot(x=range(1, 9), y=inertia_values, marker="o")
plt.title("Метод локтя для выбора количества кластеров")
plt.xlabel("Число кластеров (k)")
plt.ylabel("Инерция (сумма квадратичных отклонений внутри кластеров)")
plt.show()


# Ячейка №5: Кластеризация по возрасту и доходу + визуализация

model_age_income = KMeans(n_clusters=3, random_state=42).fit(scaled_data[:, [0, 1]])
labels_age_income = model_age_income.labels_
centers_age_income = scaler.inverse_transform(model_age_income.cluster_centers_)

plt.figure(figsize=(10, 7))
sns.scatterplot(data=df, x='возраст', y='доход_тыс_руб', hue=labels_age_income, palette='viridis')
plt.scatter(centers_age_income[:, 0], centers_age_income[:, 1], c='red', s=200, marker='x', label='Центры кластеров')
plt.scatter(df['возраст'].mean(), df['доход_тыс_руб'].mean(), color='black', s=100, marker='+', label='Общее среднее')
plt.legend(title='Кластеры')
plt.title("Диаграмма рассеяния клиентов по возрасту и доходу")
plt.show()


# Ячейка №6: Кластеризация по всем признакам и трехмерная визуализация

model_all_features = KMeans(n_clusters=3, random_state=42).fit(scaled_data)
labels_all_features = model_all_features.labels_

fig = px.scatter_3d(
    df.assign(cluster=labels_all_features),
    x='возраст',
    y='доход_тыс_руб',
    z='доля_цифровых_операций',
    color='cluster',
    hover_name='id_клиента',
    title="Интерактивное распределение клиентов по кластерам"
)
fig.update_layout(scene=dict(aspectmode="cube"))
fig.show()


# Ячейка №7: Профилирование кластеров

profiles = df.groupby(labels_all_features).agg(['mean', 'count'])
print(profiles)
```
